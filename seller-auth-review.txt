"use client";

import React, {
  createContext,
  useContext,
  useState,
  useEffect,
    useCallback,
  useMemo,
} from "react";
import { User, Address } from "@/types";
import { useToast } from "./toast-context";
import { createClient } from "@/lib/supabase/client";

interface AuthContextType {
  user: User | null;
  isAuthenticated: boolean;
  login: (email: string) => Promise<boolean>;
  register: (
    name: string,
    email: string,
    phone: string
  ) => Promise<boolean>;
  verifyOtp: (email: string, token: string) => Promise<boolean>;
  logout: () => void;
  setRole: (role: "customer" | "seller" | "admin") => void;
  addAddress: (address: Omit<Address, "id">) => void;
  updateAddress: (id: string, address: Partial<Address>) => void;
  deleteAddress: (id: string) => void;
  setDefaultAddress: (id: string) => void;
}

const AuthContext = createContext<AuthContextType | undefined>(undefined);

const ADDRESS_STORAGE_KEY = "rnl_customer_addresses_v1";

function getUserRole(
  appMetadata: Record<string, unknown> | undefined
): "customer" | "seller" | "admin" {
  const role = appMetadata?.role;

  if (role === "seller" || role === "admin") {
    return role;
  }

  return "customer";
}

export function AuthProvider({
  children,
}: {
  children: React.ReactNode;
}) {
  const [user, setUser] = useState<User | null>(null);
  const [authReady, setAuthReady] = useState(false);
  const { toast } = useToast();

  const supabase = useMemo(() => createClient(), []);

  const saveUser = useCallback((updated: User | null) => {
    setUser(updated);
  }, []);

  useEffect(() => {
    let mounted = true;

    const loadUser = async () => {
      const { data, error } = await supabase.auth.getUser();

      if (!mounted) return;

      if (error || !data.user) {
        setUser(null);
        setAuthReady(true);
        return;
      }

      const authUser = data.user;
      const storedName =
        authUser.user_metadata?.full_name ??
        authUser.user_metadata?.name ??
        "";

      let addresses: Address[] = [];

      try {
        const stored = localStorage.getItem(
          `${ADDRESS_STORAGE_KEY}:${authUser.id}`
        );
        if (stored) addresses = JSON.parse(stored) as Address[];
      } catch {
        addresses = [];
      }

      setUser({
        id: authUser.id,
        name: storedName,
        email: authUser.email ?? "",
        phone: authUser.user_metadata?.phone ?? "",
        role: getUserRole(authUser.app_metadata),
        addresses,
      });

      setAuthReady(true);
    };

    void loadUser();

    const {
  data: { subscription },
} = supabase.auth.onAuthStateChange((_event, session) => {
  if (!mounted) return;

  if (!session?.user) {
    setUser(null);
    setAuthReady(true);
    return;
  }

  const authUser = session.user;

  setUser((currentUser) => ({
    id: authUser.id,
    name:
      authUser.user_metadata?.full_name ??
      authUser.user_metadata?.name ??
      currentUser?.name ??
      "",
    email: authUser.email ?? "",
    phone: authUser.user_metadata?.phone ?? currentUser?.phone ?? "",
    role: getUserRole(authUser.app_metadata),
    addresses:
      currentUser?.id === authUser.id
        ? currentUser.addresses
        : [],
  }));

  setAuthReady(true);
});

    return () => {
      mounted = false;
      subscription.unsubscribe();
    };
  }, [supabase]);

  const login = useCallback(
    async (email: string): Promise<boolean> => {
      const normalizedEmail = email.trim().toLowerCase();

      const { error } = await supabase.auth.signInWithOtp({
        email: normalizedEmail,
        options: {
          shouldCreateUser: false,
        },
      });

      if (error) {
        toast({
          type: "error",
          title: "Unable to Send Code",
          message: error.message,
        });
        return false;
      }

      toast({
        type: "success",
        title: "Check Your Email",
        message: "We sent you a verification code.",
      });

      return true;
    },
    [supabase, toast]
  );

  const register = useCallback(
    async (
      name: string,
      email: string,
      phone: string
    ): Promise<boolean> => {
      const normalizedEmail = email.trim().toLowerCase();

      const { error } = await supabase.auth.signInWithOtp({
        email: normalizedEmail,
        options: {
          shouldCreateUser: true,
          data: {
            full_name: name.trim(),
            phone: phone.trim(),
          },
        },
      });

      if (error) {
        toast({
          type: "error",
          title: "Registration Failed",
          message: error.message,
        });
        return false;
      }

      toast({
        type: "success",
        title: "Verify Your Email",
        message: "We sent you a verification code.",
      });

      return true;
    },
    [supabase, toast]
  );

  const verifyOtp = useCallback(
    async (email: string, token: string): Promise<boolean> => {
      const { data, error } = await supabase.auth.verifyOtp({
        email: email.trim().toLowerCase(),
        token: token.trim(),
        type: "email",
      });

      if (error || !data.user) {
        toast({
          type: "error",
          title: "Verification Failed",
          message: error?.message ?? "Please request a new code.",
        });
        return false;
      }

      const authUser = data.user;

      setUser((currentUser) => ({
  id: authUser.id,
  name:
    authUser.user_metadata?.full_name ??
    authUser.user_metadata?.name ??
    currentUser?.name ??
    "",
  email: authUser.email ?? "",
  phone: authUser.user_metadata?.phone ?? currentUser?.phone ?? "",
  role: getUserRole(authUser.app_metadata),
  addresses:
    currentUser?.id === authUser.id
      ? currentUser.addresses
      : [],
}));

      toast({
        type: "success",
        title: "Email Verified",
        message: "You are now signed in.",
      });

      return true;
    },
    [supabase, toast]
  );

  const logout = useCallback(async () => {
    const { error } = await supabase.auth.signOut();

    if (error) {
      toast({
        type: "error",
        title: "Sign Out Failed",
        message: error.message,
      });
      return;
    }

    saveUser(null);

    toast({
      type: "success",
      title: "Signed Out",
      message: "You have been signed out.",
    });
  }, [supabase, saveUser, toast]);

  const persistAddresses = useCallback(
    (updated: User) => {
      saveUser(updated);

      try {
        localStorage.setItem(
          `${ADDRESS_STORAGE_KEY}:${updated.id}`,
          JSON.stringify(updated.addresses)
        );
      } catch {
        toast({
          type: "error",
          title: "Address Not Saved",
          message: "Your browser could not save this address.",
        });
      }
    },
    [saveUser, toast]
  );

  const setRole = useCallback(
    (role: "customer" | "seller" | "admin") => {
      if (role !== "customer") {
        toast({
          type: "error",
          title: "Permission Denied",
          message: "Staff roles must be assigned securely by an administrator.",
        });
      }
    },
    [toast]
  );

  const addAddress = useCallback(
    (newAddress: Omit<Address, "id">) => {
      if (!user) return;

      const address: Address = {
        ...newAddress,
        id: crypto.randomUUID(),
      };

      const existing = user.addresses.map((item) =>
        address.isDefault ? { ...item, isDefault: false } : item
      );

      const updated = {
        ...user,
        addresses: [...existing, address],
      };

      persistAddresses(updated);

      toast({
        type: "success",
        title: "Address Saved",
        message: `${address.city}, ${address.pinCode}`,
      });
    },
    [user, persistAddresses, toast]
  );

  const updateAddress = useCallback(
    (id: string, fields: Partial<Address>) => {
      if (!user) return;

      const updatedAddresses = user.addresses.map((address) => {
        if (address.id === id) return { ...address, ...fields };
        if (fields.isDefault) return { ...address, isDefault: false };
        return address;
      });

      persistAddresses({ ...user, addresses: updatedAddresses });
    },
    [user, persistAddresses]
  );

  const deleteAddress = useCallback(
    (id: string) => {
      if (!user) return;

      persistAddresses({
        ...user,
        addresses: user.addresses.filter((address) => address.id !== id),
      });

      toast({
        type: "info",
        title: "Address Deleted",
        message: "The saved address was removed.",
      });
    },
    [user, persistAddresses, toast]
  );

  const setDefaultAddress = useCallback(
    (id: string) => {
      if (!user) return;

      persistAddresses({
        ...user,
        addresses: user.addresses.map((address) => ({
          ...address,
          isDefault: address.id === id,
        })),
      });
    },
    [user, persistAddresses]
  );

  return (
    <AuthContext.Provider
      value={{
        user,
        isAuthenticated: authReady && !!user,
        login,
        register,
        verifyOtp,
        logout,
        setRole,
        addAddress,
        updateAddress,
        deleteAddress,
        setDefaultAddress,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);

  if (!context) {
    throw new Error("useAuth must be used within an AuthProvider");
  }

  return context;
}

"use client";

import React, { useState } from "react";
import Link from "next/link";
import { useRouter } from "next/navigation";
import { ArrowRight, Mail, AlertCircle, ShieldCheck } from "lucide-react";
import { StoreLayout } from "@/components/layout/StoreLayout";
import { useAuth } from "@/context/auth-context";

export default function LoginPage() {
  const [email, setEmail] = useState("");
  const [token, setToken] = useState("");
  const [step, setStep] = useState<"email" | "otp">("email");
  const [error, setError] = useState("");
  const [isSubmitting, setIsSubmitting] = useState(false);

  const { login, verifyOtp } = useAuth();
  const router = useRouter();

  const handleSendCode = async (e: React.FormEvent) => {
    e.preventDefault();
    setError("");

    const normalizedEmail = email.trim().toLowerCase();

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(normalizedEmail)) {
      setError("Please enter a valid email address.");
      return;
    }

    setIsSubmitting(true);

    try {
      const success = await login(normalizedEmail);

      if (success) {
        setEmail(normalizedEmail);
        setStep("otp");
      } else {
        setError("We couldn't send your code. Please try again.");
      }
    } catch {
      setError("Something went wrong. Please try again.");
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleVerifyCode = async (e: React.FormEvent) => {
    e.preventDefault();
    setError("");

    if (!/^\d{6}$/.test(token)) {
      setError("Enter the six-digit code from your email.");
      return;
    }

    setIsSubmitting(true);

    try {
      const success = await verifyOtp(email, token);

      if (success) {
        router.replace("/account");
      } else {
        setError("Verification failed. Check the code and try again.");
      }
    } catch {
      setError("Something went wrong. Please try again.");
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleResendCode = async () => {
    setError("");
    setIsSubmitting(true);

    try {
      const success = await login(email);

      if (success) {
        setToken("");
      } else {
        setError("We couldn't resend the code. Please try again later.");
      }
    } catch {
      setError("Unable to resend the code. Please try again.");
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <StoreLayout>
      <div className="min-h-[75vh] flex items-center justify-center py-12 px-4 sm:px-6 lg:px-8 bg-[#F5F4F0]">
        <div className="max-w-md w-full bg-white border border-neutral-200 p-8 sm:p-10 shadow-lg space-y-6">
          <div className="text-center space-y-1">
            <span className="text-[11px] uppercase tracking-widest font-mono text-neutral-500 font-bold">
              CUSTOMER ACCESS
            </span>

            <h1 className="text-2xl sm:text-3xl font-black uppercase tracking-tight font-editorial">
              {step === "email"
                ? "SIGN IN TO YOUR ACCOUNT"
                : "VERIFY YOUR EMAIL"}
            </h1>

            <p className="text-xs text-neutral-500">
              {step === "email"
                ? "Access your order tracking, saved addresses, and wishlist."
                : `Enter the six-digit code sent to ${email}.`}
            </p>
          </div>

          <div className="flex items-start gap-3 p-3 bg-neutral-100 border border-neutral-200 text-xs text-neutral-700">
            {step === "email" ? (
              <Mail className="w-4 h-4 shrink-0 mt-0.5" />
            ) : (
              <ShieldCheck className="w-4 h-4 shrink-0 mt-0.5" />
            )}

            <p>
              {step === "email"
                ? "Sign in securely using a one-time verification code sent to your email. No password required."
                : "Your account will be accessible after the verification code is confirmed."}
            </p>
          </div>

          {error && (
            <div
              role="alert"
              className="flex items-center gap-2 p-3 bg-red-50 text-red-700 text-xs border border-red-200"
            >
              <AlertCircle className="w-4 h-4 shrink-0" />
              <span>{error}</span>
            </div>
          )}

          {step === "email" ? (
            <form onSubmit={handleSendCode} className="space-y-4">
              <div className="space-y-1">
                <label
                  htmlFor="email"
                  className="text-[11px] uppercase tracking-wider font-bold text-neutral-800"
                >
                  Email Address
                </label>

                <div className="relative">
                  <input
                    id="email"
                    type="email"
                    autoComplete="email"
                    required
                    value={email}
                    onChange={(e) => setEmail(e.target.value)}
                    className="w-full pl-9 pr-3 py-2.5 text-xs border border-neutral-300 focus:border-black outline-none font-medium"
                    placeholder="name@example.com"
                  />

                  <Mail className="w-4 h-4 text-neutral-400 absolute left-3 top-3" />
                </div>
              </div>

              <button
                type="submit"
                disabled={isSubmitting}
                className="w-full py-3.5 bg-[#111111] hover:bg-black text-white text-xs uppercase tracking-widest font-black transition-colors flex items-center justify-center gap-2 shadow-sm disabled:opacity-60"
              >
                <span>
                  {isSubmitting ? "Sending Code..." : "Send Verification Code"}
                </span>
                <ArrowRight className="w-4 h-4" />
              </button>
            </form>
          ) : (
            <form onSubmit={handleVerifyCode} className="space-y-4">
              <div className="space-y-1">
                <label
                  htmlFor="otp"
                  className="text-[11px] uppercase tracking-wider font-bold text-neutral-800"
                >
                  Six-Digit Verification Code
                </label>

                <input
                  id="otp"
                  type="text"
                  inputMode="numeric"
                  autoComplete="one-time-code"
                  pattern="[0-9]{6}"
                  maxLength={6}
                  required
                  value={token}
                  onChange={(e) =>
                    setToken(e.target.value.replace(/\D/g, "").slice(0, 6))
                  }
                  className="w-full px-3 py-3 text-center text-xl tracking-[0.5em] border border-neutral-300 focus:border-black outline-none font-bold"
                  placeholder="000000"
                  aria-describedby="otp-help"
                />

                <p id="otp-help" className="text-[11px] text-neutral-500">
                  Check your inbox and spam folder.
                </p>
              </div>

              <button
                type="submit"
                disabled={isSubmitting || token.length !== 6}
                className="w-full py-3.5 bg-[#111111] hover:bg-black text-white text-xs uppercase tracking-widest font-black transition-colors flex items-center justify-center gap-2 shadow-sm disabled:opacity-60"
              >
                <span>
                  {isSubmitting ? "Verifying..." : "Verify & Sign In"}
                </span>
                <ArrowRight className="w-4 h-4" />
              </button>

              <div className="flex items-center justify-between text-xs">
                <button
                  type="button"
                  disabled={isSubmitting}
                  onClick={() => {
                    setStep("email");
                    setToken("");
                    setError("");
                  }}
                  className="text-neutral-600 underline hover:text-black disabled:opacity-60"
                >
                  Change email
                </button>

                <button
                  type="button"
                  disabled={isSubmitting}
                  onClick={handleResendCode}
                  className="font-bold text-black underline disabled:opacity-60"
                >
                  Resend code
                </button>
              </div>
            </form>
          )}

          <div className="pt-4 border-t border-neutral-200 text-center text-xs text-neutral-600">
            Don&apos;t have an account?{" "}
            <Link
              href="/register"
              className="font-bold text-black hover:underline uppercase"
            >
              Create an Account
            </Link>
          </div>
        </div>
      </div>
    </StoreLayout>
  );
}

"use client";

import React, { useState } from "react";
import Link from "next/link";

  CheckCircle2,
  Save,
  Send,
} from "lucide-react";

import { TShirtFit, TShirtSize } from "@/types";
import { slugify } from "@/lib/utils";

export default function AddProductPage() {
  const router = useRouter();
  const { toast } = useToast();

  const [name, setName] = useState("");
  const [slug, setSlug] = useState("");
  const [collection, setCollection] = useState("new-drops");
  const [tagline, setTagline] = useState("");
  const [description, setDescription] = useState("");
  const [price, setPrice] = useState("999");
  const [compareAtPrice, setCompareAtPrice] = useState("");
  const [fit, setFit] = useState<TShirtFit>("Oversized");

  );

  const [sku, setSku] = useState("");

    "S",
    "M",
    "L",
    "XL",
  ]);

  const [colors, setColors] = useState([
    { name: "Pitch Black", hex: "#111111" },
    { name: "Bone Cream", hex: "#F3EFE6" },
  ]);

    "https://images.unsplash.com/photo-1503342217505-b0a15ec3261c?auto=format&fit=crop&w=1200&q=80"
  );

    "https://images.unsplash.com/photo-1521572267360-ee0c2909d518?auto=format&fit=crop&w=1200&q=80"
  );

  const [isSubmitting, setIsSubmitting] = useState(false);
  const [statusMessage, setStatusMessage] = useState<string | null>(null);

    setName(value);
    setSlug(slugify(value));
  };

  const toggleSize = (size: TShirtSize) => {
    setSelectedSizes((previous) =>
      previous.includes(size)
        ? previous.filter((item) => item !== size)
        : [...previous, size]
    );
  };

  const handlePublish = async (asDraft = false) => {
    const productName = name.trim();
    const productSku = sku.trim();
    const productSlug = slugify(slug.trim() || productName);

    if (!productName || !productSku || !productSlug || !price.trim()) {
      toast({
        type: "error",
        title: "Missing required fields",
        message: "Enter the product name, price, and SKU.",
      });
      return;
    }

    const numericPrice = Number(price);
    const numericCompareAtPrice = compareAtPrice.trim()
      ? Number(compareAtPrice)
      : null;
    const numericGsm = Number(gsm);
    const numericStock = Number(stock);

    if (!Number.isFinite(numericPrice) || numericPrice < 0) {
      toast({
        type: "error",
        title: "Invalid price",
        message: "Enter a valid, non-negative price.",
      });
      return;
    }

    if (
      numericCompareAtPrice !== null &&
      (!Number.isFinite(numericCompareAtPrice) ||
        numericCompareAtPrice < 0)
    ) {
      toast({
        type: "error",
        title: "Invalid compare-at price",
        message: "Enter a valid price or leave this field empty.",
      });
      return;

      !Number.isInteger(numericGsm) ||

        return true;

        type: "error",
        title: "Invalid image URL",
        message: "Enter valid HTTP or HTTPS image URLs.",
      });
      return;

        title: "Missing product options",

        method: "POST",
        headers: {
          "Content-Type": "application/json",

            name: color.name.trim(),
            hex: color.hex.toUpperCase(),
          })),
          stock: numericStock,
          is_active: !asDraft,
        }),
      });

      const result = await response.json().catch(() => ({}));

        throw new Error(
          result.error || `Unable to save product (HTTP ${response.status}).`
        );
      }

      setStatusMessage(
        asDraft
          ? "Product saved successfully as a draft."
          : "Product published successfully."
      );

        type: "success",
        title: asDraft ? "Draft Saved" : "Product Published",
        message: `${productName} has been saved to your catalog.`,
      });

      router.push("/seller/products");
      router.refresh();
    } catch (error) {
      const message =
        error instanceof Error
          ? error.message
          : "Something went wrong while saving the product.";

        type: "error",
        title: "Product could not be saved",
        message,
      });

      setIsSubmitting(false);

              <span className="text-[10px] uppercase font-mono tracking-widest text-neutral-500 font-bold block mb-1">
                RIGHTNOWLIFE INTERNAL OPERATIONS
              </span>
              <h1 className="text-2xl sm:text-3xl font-black uppercase tracking-tight font-editorial">
                {title}
              </h1>
              {subtitle && <p className="text-xs text-neutral-600 mt-0.5">{subtitle}</p>}
            </div>

            {actions && <div className="flex items-center gap-2">{actions}</div>}
          </div>

          {children}
        </main>
      </div>
    </div>
  );
}

import { createServerClient } from "@supabase/ssr";
import { cookies } from "next/headers";

export async function createClient() {
  const cookieStore = await cookies();

  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!,
    {
      cookies: {
        getAll() {
          return cookieStore.getAll();
        },
        setAll(cookiesToSet) {
          try {
            cookiesToSet.forEach(({ name, value, options }) => {
              cookieStore.set(name, value, options);
            });
          } catch {
            // Cookie writes may not be available in Server Components.
            // Handle session refresh using the appropriate Next.js
            // proxy/middleware pattern.
          }
        },
      },
    }
  );
}

import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";

export async function proxy(request: NextRequest) {
  let response = NextResponse.next({ request });

  const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const supabaseKey = process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY;

  if (!supabaseUrl || !supabaseKey) {
    throw new Error("Supabase environment variables are missing.");
  }

  const supabase = createServerClient(supabaseUrl, supabaseKey, {
    cookies: {
      getAll() {
        return request.cookies.getAll();
      },
      setAll(cookiesToSet) {
        cookiesToSet.forEach(({ name, value }) => {
          request.cookies.set(name, value);
        });

        response = NextResponse.next({ request });

        cookiesToSet.forEach(({ name, value, options }) => {
          response.cookies.set(name, value, options);
        });
      },
    },
  });

  await supabase.auth.getClaims();

  return response;
}
